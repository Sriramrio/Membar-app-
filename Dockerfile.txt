
# ===================================
# BUILD STAGE
# ===================================
FROM mcr.microsoft.com/dotnet/sdk:10.0 AS build

WORKDIR /src

# Copy source code
COPY . .

# Restore NuGet packages
RUN dotnet restore backend/src/Sethu.Api/Sethu.Api.csproj

# Publish API
RUN dotnet publish backend/src/Sethu.Api/Sethu.Api.csproj \
    -c Release \
    -o /app/publish \
    /p:UseAppHost=false

# ===================================
# RUNTIME STAGE
# ===================================
FROM mcr.microsoft.com/dotnet/aspnet:10.0 AS final

WORKDIR /app

COPY --from=build /app/publish .

# Render port
ENV ASPNETCORE_URLS=http://+:10000

EXPOSE 10000

ENTRYPOINT ["dotnet", "Sethu.Api.dll"]
